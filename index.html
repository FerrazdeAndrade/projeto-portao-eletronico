📑 Resumo do Projeto

O Projeto Portão Automático é um sistema de automação residencial desenvolvido para controlar a abertura e o fechamento de um portão de garagem. A solução conecta uma interface gráfica web (HTML, CSS, JS e Python/Flask) a um circuito microcontrolado Arduino Uno R3, garantindo persistência dos dados de acionamento em banco de dados SQLite.

📋 Requisitos do Sistema

Requisitos Funcionais: Definem as funcionalidades ativas do sistema, como a capacidade de receber comandos de abertura e fechamento via controle remoto IR ou interface web, acionar o servomotor SG90 nos ângulos programados, registrar histórico de eventos no banco de dados SQLite e sinalizar o status em tempo real por LEDs e Buzzer.

Requisitos Não Funcionais: Determinam a qualidade e desempenho da aplicação, como o tempo de resposta da animação na interface (máximo de 3 segundos), usabilidade e responsividade do layout CSS, confiabilidade na transmissão de dados via porta Serial a 9600 bps e geração de áudio no navegador via Web Audio API.

RF01 - Controle do Servo: Movimentação angular precisa de 0° a 120°.

RF02 - Receptor IR: Leitura de frequência infravermelha para controle remoto físico.

RF03 - Auditoria de BD: Incremento automático de contador a cada ciclo no banco SQLite.

RNF01 - Usabilidade: Layout proporcional expandido para visualização clara da garagem.

🤖 ARDUINO E SUAS FUNÇÕES

Servo Motor SG90: Atua como o mecanismo mecânico de elevação do portão. Quando a ação de abertura é disparada, o motor executa a rotação de 0° até 120° (posição aberto). Para fechar, o motor retorna o curso de 120° até 0° (posição fechado).

LEDs e Buzzer Sonoro: O sistema conta com dois LEDs indicadores de estado. Durante o processo de subida, o LED Verde pisca de forma sincronizada com os bipes da Buzzer. Durante a descida, o LED Vermelho pisca em conjunto com o alarme sonoro, garantindo alertas visuais e auditivos de segurança.

Sensor Infravermelho (IR): Fica encarregado da recepção dos sinais sem fio enviados pelo controle remoto portátil, permitindo acionar a abertura e o fechamento à distância. 

#include <Servo.h>
#include <IRremote.h>

// Instância das bibliotecas
Servo meuServo;

// Definição dos Pinos
const int RECV_PIN     = 2;
const int LED_VERMELHO = 4;
const int LED_VERDE    = 5;
const int SERVO_PIN    = 9;
const int PIEZO_BUZZER = 10;

// Código Hexadecimal do Botão do Controle IR
const uint16_t COMANDO_BOTAO = 0x44;

// Estado do Portão: 0 = Fechado, 1 = Aberto
int estadoPortao = 0;

void setup() {

    Serial.begin(9600);

    // Configuração dos Pinos como Saída
    pinMode(LED_VERMELHO, OUTPUT);
    pinMode(LED_VERDE, OUTPUT);
    pinMode(PIEZO_BUZZER, OUTPUT);

    // Inicialização do Servo no ângulo 0° (Fechado)
    meuServo.attach(SERVO_PIN);
    meuServo.write(0);

    // Estado Inicial dos LEDs
    digitalWrite(LED_VERMELHO, HIGH);
    digitalWrite(LED_VERDE, LOW);
    digitalWrite(PIEZO_BUZZER, LOW);

    // Inicialização do Receptor IR
    IrReceiver.begin(RECV_PIN, ENABLE_LED_FEEDBACK);

    Serial.println("Sistema de Portao Automatico Iniciado!");
}

void loop() {

    // 1. LEITURA VIA CONTROLE REMOTO IR
    if (IrReceiver.decode()) {

        uint16_t comando = IrReceiver.decodedIRData.command;

        if (comando == COMANDO_BOTAO) {

            alternarPortao();

        } else if (comando != 0 && comando != 0xFFFF) {

            Serial.print("Botao ignorado: 0x");
            Serial.println(comando, HEX);
        }

        IrReceiver.resume();
    }


    // 2. LEITURA VIA COMANDO SERIAL (PYTHON/FLASK)
    if (Serial.available() > 0) {

        char comandoSerial = Serial.read();

        if (comandoSerial == 'A' && estadoPortao == 0) {

            alternarPortao(); // Abre

        } else if (comandoSerial == 'F' && estadoPortao == 1) {

            alternarPortao(); // Fecha
        }
    }
}


// Função para controlar a movimentação, LEDs e Buzzer
void alternarPortao() {

    if (estadoPortao == 0) {

        // --- ABRIR PORTÃO (0° -> 120°) ---

        for (int angulo = 0; angulo <= 120; angulo += 10) {

            meuServo.write(angulo);

            digitalWrite(LED_VERDE, HIGH);
            digitalWrite(LED_VERMELHO, LOW);

            tocarBuzzer(100);

            delay(80);

            digitalWrite(LED_VERDE, LOW);
        }

        digitalWrite(LED_VERDE, HIGH);

        estadoPortao = 1;

        Serial.println("Estado: ABERTO");

    } else {

        // --- FECHAR PORTÃO (120° -> 0°) ---

        for (int angulo = 120; angulo >= 0; angulo -= 10) {

            meuServo.write(angulo);

            digitalWrite(LED_VERMELHO, HIGH);
            digitalWrite(LED_VERDE, LOW);

            tocarBuzzer(100);

            delay(80);

            digitalWrite(LED_VERMELHO, LOW);
        }

        digitalWrite(LED_VERMELHO, HIGH);

        estadoPortao = 0;

        Serial.println("Estado: FECHADO");
    }
}


void tocarBuzzer(int duracao) {

    digitalWrite(PIEZO_BUZZER, HIGH);

    delay(duracao);

    digitalWrite(PIEZO_BUZZER, LOW);
}
